 V2 智能检索与核心概念优化计划

 背景与目标

 基于 v2-descriptions.md 的需求，实现以下功能：

 1. Embedding 向量化: 对分片进行 embedding，使用 mlx-community/Qwen3-Embedding-0.6B-4bit-DWQ 模型
 2. 混合检索系统: 点击 graph 节点时，支持 向量检索 + reranker + BM25 + RRF 召回
 3. Source-Grounded 总结: 使用大模型基于召回分片进行总结阐述
 4. 改进 Core Concepts: 基于社区发现算法（参考 LiteRAG）改进核心概念提取质量

 impl HybridRetriever {
     /// 基于节点名称/描述进行混合检索
     pub async fn retrieve_for_node(
         &self,
         node_id: &str,
         top_k: usize,
     ) -> Result<Vec<RetrievalResult>, String> {
         // 1. 获取节点信息
         let node = self.get_node_info(node_id).await?;

         // 2. 向量检索 (Top 50)
         let vector_results = self.vector_search(&node.name, &node.description, 50).await?;

         // 3. BM25 检索 (Top 50)
         let bm25_results = self.bm25_search(&node.name, &node.description, 50).await?;

         // 4. RRF 融合 (Reciprocal Rank Fusion)
         let fused_results = self.reciprocal_rank_fusion(
             &vector_results,
             &bm25_results,
             60.0  // RRF 参数 k
         );

         // 5. Reranker 重排序 (Top 20)
         let reranked = self.rerank(&node, &fused_results[..20.min(fused_results.len())]).await?;

         // 6. 返回 Top K
         Ok(reranked.into_iter().take(top_k).collect())
     }

     /// 向量相似度搜索
     async fn vector_search(
         &self,
         node_name: &str,
         node_description: &str,
         top_k: usize,
     ) -> Result<Vec<SearchResult>, String> {
         // 1. 生成查询 embedding
         let query_embedding = self.embed_query(&format!("{} {}", node_name, node_description)).await?;

         // 2. 计算余弦相似度 (使用 SQLite 或内存计算)
         // 由于 SQLite 不支持向量运算，在 Rust 中加载并计算
         let embeddings = self.get_all_embeddings_for_book(book_id).await?;
         let mut scores: Vec<(String, f64)> = embeddings
             .iter()
             .map(|(chunk_id, emb)| {
                 let similarity = cosine_similarity(&query_embedding, emb);
                 (chunk_id.clone(), similarity as f64)
             })
             .collect();

         Ok(scores.into_iter().take(top_k).collect())

 #[derive(Serialize)]
 struct RetrievalResponse {
     chunks: Vec<RetrievalResult>,
     total_found: usize,
 }

 #[derive(Serialize)]
 struct SummaryResponse {
     summary: String,
     citations: Vec<CitationResponse>,
     sources: Vec<ChunkSummary>,
 }

 // 处理函数
 async fn node_retrieval(
     Path(node_id): Path<String>,
     Query(params): Query<RetrievalParams>,
     pool: axum::extract::State<SqlitePool>,
 ) -> Result<Json<RetrievalResponse>, AppError> {
     let retriever = HybridRetriever::new(pool.clone());
     let top_k = params.top_k.unwrap_or(10);
     let results = retriever.retrieve_for_node(&node_id, top_k).await?;

     Ok(Json(RetrievalResponse {
         chunks: results,
         total_found: results.len(),
     }))
 }

 async fn node_summary(
     Path(node_id): Path<String>,
     pool: axum::extract::State<SqlitePool>,
 ) -> Result<Json<SummaryResponse>, AppError> {
     // 1. 检索相关 chunks
     let retriever = HybridRetriever::new(pool.clone());
     let retrieval_results = retriever.retrieve_for_node(&node_id, 10).await?;

     // 2. 获取节点信息
     let node = get_node_info(&pool, &node_id).await?;

     // 3. 生成总结
     let llm = LlmClient::new(&llm_api_url);
     let summary = llm.generate_source_grounded_summary(
         SourceGroundedSummaryRequest {
             node_name: node.name,
             node_description: node.description,
             retrieval_results,
             language: node.language,
         }
     ).await?;

     Ok(Json(SummaryResponse {
         summary: summary.summary,
         citations: summary.citations,
         sources: retrieval_results.into_iter().map(|r| ...).collect(),
     }))
 }
